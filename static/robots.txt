# to crawl them.

# Reference: https://developers.google.com/search/docs/crawling-indexing/robots/robots_txt

Disallow: /static/fonts/
Disallow: /static/images/
Disallow: /static/pages/
Disallow: /static/scripts/
Disallow: /static/vendor/

User-agent: Googlebot
Allow: /static/blocks/
Allow: /static/fonts/
Allow: /static/images/
Allow: /static/pages/
Allow: /static/scripts/
Allow: /static/vendor/
